function getDataByPrefix(prefix){

    const elements = document.querySelectorAll(
        `[name^="${prefix}"]`
    );

    const result = {};

    elements.forEach(el => {

        const keys = el.name
            .replace(/\]/g, "")
            .split("[");

        let current = result;

        keys.forEach((key, index) => {

            const isLast = index === keys.length - 1;

            if(isLast){

                current[key] = el.value;

            }else{

                if(current[key] === undefined){

                    const nextKey = keys[index + 1];

                    current[key] = isNaN(nextKey)
                        ? {}
                        : [];
                }

                current = current[key];
            }

        });

    });

    return result;
}
